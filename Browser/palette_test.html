/// Configures palette luminance and contrast
                        thresholds
                    #[derive(Component, Debug, Clone)]
                    pub struct ThemeConfig<'a> {
                    pub contrast_ratio: f32, // Target contrast factor
                    pub profile_name: &'a str,
                    }

                    impl<'a> ThemeConfig<'a> {
                     pub fn validate_palette(&self, luminance: f32) -> RenderState {
                    info!("Loading profile: {:?}\n", self.profile_name);
                    if self.contrast_ratio >= 4.5 && luminance > 0.0 {
                    RenderState::Active(true)
                    } else {
                    RenderState::Fallback
                    }
                    }
                    }
